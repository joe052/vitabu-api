# Base image
FROM node:24-alpine

# Set working directory
WORKDIR /app

# Copy package.json and package-lock.json
COPY package*.json ./
RUN npm ci

# Copy the rest of the application code
COPY . .
RUN npm run build

# Environment variables
ENV NODE_ENV=production

# Expose the port the app runs on
EXPOSE 4000

# Start the application
CMD ["node", "dist/main"]
